`timescale 1ns / 1ps

module tb_Gowin_rPLL();

    // 1. Declaración de estímulos y salidas
    reg clkin;
    wire clkout;

    // 2. Instancia del Device Under Test (DUT)
    Gowin_rPLL dut (
        .clkout(clkout),
        .clkin(clkin)
    );

    // 3. Generación del reloj de entrada (27 MHz)
    // Frecuencia = 27 MHz -> Periodo = 1 / 27,000,000 = 37.037 ns
    // Alternamos el estado de la señal cada 18.518 ns.
    initial begin
        clkin = 0;
        forever #18.518 clkin = ~clkin;
    end

    // 4. Secuencia principal de simulación
    initial begin
        // Configuración para guardar las formas de onda
        $dumpfile("sim_pll.vcd");
        $dumpvars(0, tb_Gowin_rPLL);

        // Los modelos de simulación de los PLL suelen requerir algo de tiempo
        // (tiempo de lock/enganche) antes de empezar a emitir un reloj estable.
        // Simulamos durante 10 microsegundos.
        #10000; 

        $display("Simulación finalizada.");
        $finish;
    end

    // 5. Bloque auxiliar para verificar la frecuencia generada
    // Mide el tiempo entre dos flancos de subida consecutivos de clkout
    time t1, t2;
    initial begin
        // Esperamos a que el PLL arranque y estabilice (ignoramos los primeros flancos)
        #1000; 
        @(posedge clkout);
        t1 = $realtime;
        @(posedge clkout);
        t2 = $realtime;
        
        $display("--------------------------------------------------");
        $display("Periodo de clkin (Entrada)  : 37.037 ns (27 MHz)");
        $display("Periodo de clkout detectado : %f ns", t2 - t1);
        $display("Frecuencia de salida aprox  : %f MHz", 1000.0 / (t2 - t1));
        $display("--------------------------------------------------");
    end

endmodule
