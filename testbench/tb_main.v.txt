`timescale 1ns / 1ps

module tb_top_module();

    // 1. Declaración de señales
    // Los inputs del top_module se declaran como 'reg' para poder asignarles valores.
    reg clk_27m;
    reg rst_n;
    
    // Los outputs del top_module se declaran como 'wire' para observar su estado.
    wire O_tmds_clk_p;
    wire O_tmds_clk_n;
    wire [2:0] O_tmds_data_p;
    wire [2:0] O_tmds_data_n;

    // 2. Instancia del módulo top (Device Under Test - DUT)
    top_module dut (
        .clk_27m(clk_27m),
        .rst_n(rst_n),
        .O_tmds_clk_p(O_tmds_clk_p),
        .O_tmds_clk_n(O_tmds_clk_n),
        .O_tmds_data_p(O_tmds_data_p),
        .O_tmds_data_n(O_tmds_data_n)
    );

    // 3. Generación del reloj de 27 MHz
    // Frecuencia = 27 MHz -> Periodo = 1 / 27e6 = 37.037 ns.
    // Invertimos la señal cada 18.518 ns (mitad del periodo).
    initial begin
        clk_27m = 0;
        forever #18.518 clk_27m = ~clk_27m;
    end

    // 4. Secuencia de estímulos (Reset y tiempo de simulación)
    initial begin
        // Configuración para visualizar ondas en herramientas como GTKWave o ModelSim
        $dumpfile("sim_top_module.vcd");
        $dumpvars(0, tb_top_module);

        // Aplicamos reset activo en bajo al inicio
        rst_n = 0;
        
        // Esperamos 100 ns para estabilizar
        #100;
        
        // Liberamos el reset
        rst_n = 1;

        // Tiempo de simulación. 
        // ¡Ojo aquí! Transmitir un solo frame VGA/HDMI completo a 60Hz toma ~16.6 milisegundos.
        // Si quieres ver una pantalla completa simulada, necesitas simular por mucho tiempo.
        // Aquí simulamos 2 milisegundos para ver cómo se establecen los relojes y los primeros píxeles.
        #2000000; 
        
        $display("Simulación terminada con éxito.");
        $finish;
    end

endmodule
