`timescale 1ns / 1ps

module tb_vga_controller;

    // Entradas
    reg clk_pixel;
    reg rst_n;

    // Salidas
    wire hs;
    wire vs;
    wire de;
    wire [7:0] r;
    wire [7:0] g;
    wire [7:0] b;

    // Instancia del módulo
    vga_controller uut (
        .clk_pixel(clk_pixel),
        .rst_n(rst_n),
        .hs(hs),
        .vs(vs),
        .de(de),
        .r(r),
        .g(g),
        .b(b)
    );

    // Generador del reloj de 25 MHz
    initial begin
        clk_pixel = 0;

        forever begin
            #20;
            clk_pixel = ~clk_pixel;
        end
    end

    // Prueba del reset
    initial begin

        // Guardar las señales para EPWave
        $dumpfile("dump.vcd");
        $dumpvars(0, tb_vga_controller);

        // Activar reset
        rst_n = 0;

        // Esperar 100 ns
        #100;

        // Desactivar reset
        rst_n = 1;

        // Ejecutar la simulación
        #20000000;

        // Finalizar
        $finish;

    end

endmodule
