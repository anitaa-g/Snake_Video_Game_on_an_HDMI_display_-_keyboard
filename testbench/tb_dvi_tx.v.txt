`timescale 1ns / 1ps

module tb_DVI_TX_Top();

    // 1. Declaración de señales de entrada (reg)
    reg I_rst_n;
    reg I_serial_clk;
    reg I_rgb_clk;
    reg I_rgb_vs;
    reg I_rgb_hs;
    reg I_rgb_de;
    reg [7:0] I_rgb_r;
    reg [7:0] I_rgb_g;
    reg [7:0] I_rgb_b;

    // 2. Declaración de señales de salida (wire)
    wire O_tmds_clk_p;
    wire O_tmds_clk_n;
    wire [2:0] O_tmds_data_p;
    wire [2:0] O_tmds_data_n;

    // 3. Instancia del módulo DVI (Device Under Test)
    DVI_TX_Top dut (
        .I_rst_n(I_rst_n),
        .I_serial_clk(I_serial_clk),
        .I_rgb_clk(I_rgb_clk),
        .I_rgb_vs(I_rgb_vs),
        .I_rgb_hs(I_rgb_hs),
        .I_rgb_de(I_rgb_de),
        .I_rgb_r(I_rgb_r),
        .I_rgb_g(I_rgb_g),
        .I_rgb_b(I_rgb_b),
        .O_tmds_clk_p(O_tmds_clk_p),
        .O_tmds_clk_n(O_tmds_clk_n),
        .O_tmds_data_p(O_tmds_data_p),
        .O_tmds_data_n(O_tmds_data_n)
    );

    // 4. Generación del Reloj Serial (~125 MHz)
    // Periodo de 8 ns (Alterna cada 4 ns)
    initial begin
        I_serial_clk = 0;
        forever #4 I_serial_clk = ~I_serial_clk;
    end

    // 5. Generación del Reloj de Píxeles RGB (~25 MHz)
    // Periodo de 40 ns (Alterna cada 20 ns)
    // Nota: En un diseño real, estos dos relojes deben estar sincronizados en fase por el PLL
    initial begin
        I_rgb_clk = 0;
        forever #20 I_rgb_clk = ~I_rgb_clk;
    end

    // 6. Secuencia de estímulos principal
    initial begin
        // Archivo para ver las ondas en un visor (GTKWave, ModelSim, etc.)
        $dumpfile("sim_dvi_tx.vcd");
        $dumpvars(0, tb_DVI_TX_Top);

        // A. Inicialización y Reset
        I_rst_n = 0;
        I_rgb_vs = 0;
        I_rgb_hs = 0;
        I_rgb_de = 0; // Data Enable en 0 (Fuera de la zona visible de pantalla)
        I_rgb_r = 8'h00;
        I_rgb_g = 8'h00;
        I_rgb_b = 8'h00;

        #100;
        I_rst_n = 1; // Liberamos el reset
        #200;

        // B. Simulamos el inicio de una línea visible en la pantalla
        // Sincronizamos con el flanco de subida del reloj de píxeles
        @(posedge I_rgb_clk);
        I_rgb_de = 1;           // Indicamos que estamos enviando píxeles válidos
        I_rgb_r = 8'hFF;        // Mandamos un píxel completamente ROJO (R=255, G=0, B=0)
        I_rgb_g = 8'h00;
        I_rgb_b = 8'h00;

        // C. Siguiente ciclo de reloj: otro píxel
        @(posedge I_rgb_clk);
        I_rgb_r = 8'h00;        // Mandamos un píxel completamente VERDE
        I_rgb_g = 8'hFF;
        I_rgb_b = 8'h00;

        // D. Terminamos la zona visible
        @(posedge I_rgb_clk);
        I_rgb_de = 0;           // Apagamos el Data Enable
        I_rgb_r = 8'h00;
        I_rgb_g = 8'h00;
        I_rgb_b = 8'h00;

        // Dejamos correr un poco más de tiempo para observar la serialización
        #1000;
        
        $display("Simulación de DVI completada.");
        $finish;
    end

endmodule
